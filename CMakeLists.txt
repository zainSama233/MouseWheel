cmake_minimum_required(VERSION 3.25)
project(MouseWheel VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
add_compile_definitions(QT_NO_KEYWORDS)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/toolchain.json" TOOLCHAIN_JSON)
string(JSON QT_VERSION GET "${TOOLCHAIN_JSON}" qt)
find_package(Qt6 ${QT_VERSION} EXACT REQUIRED COMPONENTS Core Gui Widgets Test)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive-)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
endif()
add_library(wheel_core src/core/model.cpp src/core/interaction.cpp src/config/config_store.cpp src/tools/annotation_document.cpp)
target_include_directories(wheel_core PUBLIC src)
target_link_libraries(wheel_core PUBLIC Qt6::Core Qt6::Gui)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE wheel_core Qt6::Test)
  add_test(NAME core COMMAND core_tests)
  add_executable(config_tests tests/config_tests.cpp)
  target_link_libraries(config_tests PRIVATE wheel_core Qt6::Test)
  add_test(NAME config COMMAND config_tests)
  add_executable(annotation_tests tests/annotation_tests.cpp)
  target_link_libraries(annotation_tests PRIVATE wheel_core Qt6::Test Qt6::Widgets)
  add_test(NAME annotation COMMAND annotation_tests)
endif()
option(BUILD_APP "Build Windows desktop app" ON)
if(BUILD_APP)
  if(NOT WIN32)
    message(FATAL_ERROR "The native backend currently requires Windows. See docs/validation.md.")
  endif()
  add_library(wheel_platform src/platform/windows_input.cpp src/platform/input_service.h src/platform/windows_injection.cpp)
  target_link_libraries(wheel_platform PUBLIC wheel_core Qt6::Widgets user32 shcore wtsapi32)
  add_library(wheel_ui src/ui/wheel_window.cpp src/ui/settings_window.cpp src/ui/theme.cpp src/tools/pinned_image.cpp src/tools/screen_annotation_session.cpp src/tools/screenshot_session.cpp)
  target_link_libraries(wheel_ui PUBLIC wheel_core Qt6::Widgets user32 dwmapi)
  add_executable(MouseWheel WIN32 src/main.cpp src/app.cpp)
  target_link_libraries(MouseWheel PRIVATE wheel_platform wheel_ui)
  target_compile_definitions(MouseWheel PRIVATE MOUSEWHEEL_VERSION="${PROJECT_VERSION}")
  install(TARGETS MouseWheel RUNTIME DESTINATION .)
  if(BUILD_TESTING)
    add_executable(injection_tests tests/injection_tests.cpp)
    target_link_libraries(injection_tests PRIVATE wheel_platform Qt6::Test)
    add_test(NAME injection COMMAND injection_tests)
    add_executable(ui_tests tests/ui_tests.cpp)
    target_link_libraries(ui_tests PRIVATE wheel_ui Qt6::Test)
    add_test(NAME ui COMMAND ui_tests)
    add_executable(desktop_tests tests/desktop_tests.cpp)
    target_link_libraries(desktop_tests PRIVATE wheel_platform wheel_ui Qt6::Test)
    add_test(NAME desktop COMMAND desktop_tests)
    add_executable(app_tests tests/app_tests.cpp src/app.cpp)
    target_link_libraries(app_tests PRIVATE wheel_platform wheel_ui Qt6::Test)
    add_test(NAME app COMMAND app_tests)
    set_tests_properties(app PROPERTIES LABELS desktop RUN_SERIAL TRUE TIMEOUT 60)
    add_executable(screen_annotation_tests tests/screen_annotation_tests.cpp)
    target_link_libraries(screen_annotation_tests PRIVATE wheel_ui Qt6::Test)
    add_test(NAME screen_annotation COMMAND screen_annotation_tests)
    set_tests_properties(screen_annotation PROPERTIES LABELS desktop RUN_SERIAL TRUE TIMEOUT 60)
    add_executable(screenshot_tests tests/screenshot_tests.cpp)
    target_link_libraries(screenshot_tests PRIVATE wheel_ui Qt6::Test)
    add_test(NAME screenshot COMMAND screenshot_tests)
    set_tests_properties(screenshot PROPERTIES LABELS desktop RUN_SERIAL TRUE TIMEOUT 60)
    set_tests_properties(desktop PROPERTIES LABELS desktop RUN_SERIAL TRUE TIMEOUT 90)
  endif()
endif()
