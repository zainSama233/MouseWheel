cmake_minimum_required(VERSION 3.25)
project(MouseWheel VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
add_compile_definitions(QT_NO_KEYWORDS)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/toolchain.json" TOOLCHAIN_JSON)
string(JSON QT_VERSION GET "${TOOLCHAIN_JSON}" qt)
find_package(Qt6 ${QT_VERSION} EXACT REQUIRED COMPONENTS Core Gui Widgets Test)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive-)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
endif()
add_library(wheel_core src/core/model.cpp src/core/image_asset.cpp src/core/interaction.cpp src/config/config_store.cpp src/config/action_codec.cpp src/tools/annotation_document.cpp)
target_include_directories(wheel_core PUBLIC src)
target_link_libraries(wheel_core PUBLIC Qt6::Core Qt6::Gui)
include(CTest)
if(BUILD_TESTING)
  add_executable(action_tests tests/action_tests.cpp)
  target_link_libraries(action_tests PRIVATE wheel_core Qt6::Test)
  add_test(NAME actions COMMAND action_tests)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE wheel_core Qt6::Test)
  add_test(NAME core COMMAND core_tests)
  add_executable(config_tests tests/config_tests.cpp)
  target_link_libraries(config_tests PRIVATE wheel_core Qt6::Test)
  add_test(NAME config COMMAND config_tests)
  add_executable(annotation_tests tests/annotation_tests.cpp)
  target_link_libraries(annotation_tests PRIVATE wheel_core Qt6::Test Qt6::Widgets)
  add_test(NAME annotation COMMAND annotation_tests)
endif()
option(BUILD_APP "Build Windows desktop app" ON)
if(BUILD_APP)
  if(NOT WIN32)
    message(FATAL_ERROR "The native backend currently requires Windows. See docs/validation.md.")
  endif()
  add_library(wheel_platform src/platform/windows_input.cpp src/platform/window_context.cpp src/platform/program_icon.cpp src/platform/application_catalog.cpp src/platform/input_service.h src/platform/windows_injection.cpp src/platform/desktop_actions.cpp src/platform/shortcut_capture.cpp)
  target_link_libraries(wheel_platform PUBLIC wheel_core Qt6::Widgets user32 shcore wtsapi32 dwmapi shell32 ole32 uuid advapi32)
  find_package(Qt6 ${QT_VERSION} EXACT REQUIRED COMPONENTS Svg Network Concurrent)
  add_library(wheel_ui src/ui/action_icons.cpp src/ui/application_picker.cpp src/ui/trigger_rules_editor.cpp src/ui/icons/icons.qrc src/tools/launcher.cpp src/tools/website_icon.cpp src/ui/wheel_window.cpp src/ui/settings_window.cpp src/ui/slot_editor.cpp src/ui/shortcut_editor.cpp src/ui/theme.cpp src/tools/pinned_image.cpp src/tools/screen_annotation_session.cpp src/tools/screenshot_session.cpp src/tools/region_capture.cpp src/tools/ocr_session.cpp src/tools/local_ocr.cpp)
  target_link_libraries(wheel_ui PUBLIC wheel_core wheel_platform Qt6::Widgets Qt6::Svg Qt6::Network Qt6::Concurrent windowsapp user32 dwmapi shell32 ole32 oleaut32 uuid advapi32)
  add_executable(MouseWheel WIN32 src/main.cpp src/app.cpp)
  target_link_libraries(MouseWheel PRIVATE wheel_platform wheel_ui)
  target_compile_definitions(MouseWheel PRIVATE MOUSEWHEEL_VERSION="${PROJECT_VERSION}")
  install(TARGETS MouseWheel RUNTIME DESTINATION .)
  if(BUILD_TESTING)
    add_executable(iteration_tests tests/iteration_tests.cpp)
    target_link_libraries(iteration_tests PRIVATE wheel_ui Qt6::Test)
    add_test(NAME iteration COMMAND iteration_tests -o "${CMAKE_CURRENT_BINARY_DIR}/iteration-results.txt,txt")
    add_executable(icon_tests tests/icon_tests.cpp)
    target_link_libraries(icon_tests PRIVATE wheel_ui Qt6::Test)
    add_test(NAME icons COMMAND icon_tests)
    add_executable(discovery_tests tests/discovery_tests.cpp)
    target_link_libraries(discovery_tests PRIVATE wheel_ui Qt6::Test)
    add_test(NAME discovery COMMAND discovery_tests)
    set_tests_properties(discovery PROPERTIES LABELS desktop RUN_SERIAL TRUE TIMEOUT 60)
    add_executable(launch_probe tests/launch_probe.cpp)
    target_link_libraries(launch_probe PRIVATE Qt6::Core)
    add_executable(extended_tests tests/extended_tests.cpp)
    target_link_libraries(extended_tests PRIVATE wheel_ui Qt6::Test)
    add_dependencies(extended_tests launch_probe)
    add_test(NAME extended COMMAND extended_tests)
    set_tests_properties(extended PROPERTIES LABELS desktop RUN_SERIAL TRUE TIMEOUT 90)
    add_executable(launcher_tests tests/launcher_tests.cpp)
    target_link_libraries(launcher_tests PRIVATE wheel_ui Qt6::Test ole32 uuid)
    add_dependencies(launcher_tests launch_probe)
    add_test(NAME launcher COMMAND launcher_tests)
    add_executable(injection_tests tests/injection_tests.cpp)
    target_link_libraries(injection_tests PRIVATE wheel_platform Qt6::Test)
    add_test(NAME injection COMMAND injection_tests)
    add_executable(ui_tests tests/ui_tests.cpp)
    target_link_libraries(ui_tests PRIVATE wheel_ui Qt6::Test)
    add_test(NAME ui COMMAND ui_tests)
    add_executable(desktop_tests tests/desktop_tests.cpp)
    target_link_libraries(desktop_tests PRIVATE wheel_platform wheel_ui Qt6::Test)
    add_test(NAME desktop COMMAND desktop_tests -o "${CMAKE_CURRENT_BINARY_DIR}/desktop-results.txt,txt")
    add_executable(app_tests tests/app_tests.cpp src/app.cpp)
    target_link_libraries(app_tests PRIVATE wheel_platform wheel_ui Qt6::Test)
    add_test(NAME app COMMAND app_tests)
    set_tests_properties(app PROPERTIES LABELS desktop RUN_SERIAL TRUE TIMEOUT 60)
    add_executable(screen_annotation_tests tests/screen_annotation_tests.cpp)
    target_link_libraries(screen_annotation_tests PRIVATE wheel_ui Qt6::Test)
    add_test(NAME screen_annotation COMMAND screen_annotation_tests -o "${CMAKE_CURRENT_BINARY_DIR}/screen-annotation-results.txt,txt")
    set_tests_properties(screen_annotation PROPERTIES LABELS desktop RUN_SERIAL TRUE TIMEOUT 60)
    add_executable(screenshot_tests tests/screenshot_tests.cpp)
    target_link_libraries(screenshot_tests PRIVATE wheel_ui Qt6::Test)
    add_test(NAME screenshot COMMAND screenshot_tests)
    set_tests_properties(screenshot PROPERTIES LABELS desktop RUN_SERIAL TRUE TIMEOUT 60)
    set_tests_properties(desktop PROPERTIES LABELS desktop RUN_SERIAL TRUE TIMEOUT 90)
  endif()
endif()
