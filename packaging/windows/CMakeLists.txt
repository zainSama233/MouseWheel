cmake_minimum_required(VERSION 3.25)
project(MouseWheelPortable LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT EXISTS "${PAYLOAD}")
  message(FATAL_ERROR "Supply PAYLOAD cabinet")
endif()
file(SHA256 "${PAYLOAD}" PAYLOAD_ID)
configure_file(payload.rc.in payload.rc @ONLY)
configure_file(payload.h.in payload.h @ONLY)
add_executable(MouseWheelPortable WIN32 launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")
target_include_directories(MouseWheelPortable PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(MouseWheelPortable PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(MouseWheelPortable PRIVATE /W4 /utf-8)
target_link_libraries(MouseWheelPortable PRIVATE setupapi shell32)
